```jsx
async function parseJson(response) {
  const text = await response.text();

  if (!text) {
    return null;
  }

  try {
    return JSON.parse(text);
  } catch (error) {
    const responsePreview = text
      .replace(/\s+/g, ' ')
      .slice(0, 150);

    throw new Error(
      `서버 응답 형식이 올바르지 않습니다.`
      + ` HTTP ${response.status}`
      + (
        responsePreview
          ? `: ${responsePreview}`
          : ''
      )
    );
  }
}
```

json 응답 오류는 전체적으로 냅둘 예정입니다. 나중에 ai에게 오류를 반환하는 react 식 방법을 물어볼 예정입니다.

```jsx
function createFetchOptions(method, body, accessToken) {
  const fetchOptions = {
    method,
    headers: {},
    credentials: 'include'
  };

  if (body instanceof FormData) {
    fetchOptions.body = body;
  } else if (body !== undefined) {
    fetchOptions.headers['Content-Type'] =
      'application/json';

    fetchOptions.body = JSON.stringify(body);
  }

  if (accessToken) {
    fetchOptions.headers.Authorization =
      `Bearer ${accessToken}`;
  }

  return fetchOptions;
}
```

json이나 파일 요청, 인증 헤더를 추가하는 방식은 그대로 냅둘 예정입니다. 나중에 ai에게 react 식으로 마이그레이션 방법을 물어보고 적용할 계획입니다.

```jsx

}
```

path를 일일히 다 확인해서 지저분한 느낌이라 react 마이그레이션 하면서 바꿔볼 예정입니다.

```jsx
async function performRefreshAccessToken() {
  const response = await fetch(
    `${API_BASE_URL}/users/token/refresh`,
    {
      method: 'POST',
      credentials: 'include'
    }
  );

  const result = await parseJson(response);

  if (!response.ok) {
    clearAuth();

    throw new Error(
      result?.message || '로그인이 만료되었습니다.'
    );
  }

  const newAccessToken =
    result?.data?.accessToken;

  if (!newAccessToken) {
    clearAuth();

    throw new Error(
      '토큰 재발급에 실패했습니다.'
    );
  }

  localStorage.setItem(
    'accessToken',
    newAccessToken
  );

  return newAccessToken;
}
```

토큰 재발행 함수입니다. JSON 형태로 응답 받고 만약 재발급에 실패하면 clearAuth(로그아웃) 합니다. clearAuth 에서 state로 user를 관리하기 떄문에 화면이 재렌더링 되면서 로그아웃 상태로 표시 될 것 입니다.

여기서 어떻게 다른 파일로 분리시킨 clearAuth를 가져오고 실행 시키지? 생각했는데 react에서 handleUnauthorezed를 통해 clearAuth에서 useEffect로 선언된 setUnauthorizedHandler를 실행시켜 clearAuth를 실행시키면 되는 것 이었습니다.

```jsx
function createProfileMenu(header, profileButton) {
  const profileMenu = document.createElement('div');
  profileMenu.className = 'profile-menu';

  profileMenu.innerHTML = `
    <button type="button" id="userEditMenu">회원정보수정</button>
    <button type="button" id="passwordEditMenu">비밀번호수정</button>
    <button type="button" id="logoutMenu">로그아웃</button>
  `;

  header.appendChild(profileMenu);

  profileButton.addEventListener('click', function () {
    const opened = profileMenu.classList.toggle('is-open');
    profileButton.setAttribute('aria-expanded', String(opened));
  });

  profileMenu
    .querySelector('#userEditMenu')
    .addEventListener('click', function () {
      location.href = '../userEdit/index.html';
    });

  profileMenu
    .querySelector('#passwordEditMenu')
    .addEventListener('click', function () {
      location.href = '../passwordEdit/index.html';
    });

  profileMenu
    .querySelector('#logoutMenu')
    .addEventListener('click', function () {
      localStorage.clear();
      location.href = '../home/index.html';
    });

  document.addEventListener('click', function (event) {
    if (
      !profileButton.contains(event.target)
      && !profileMenu.contains(event.target)
    ) {
      profileMenu.classList.remove('is-open');
      profileButton.setAttribute('aria-expanded', 'false');
    }
  });
}
```

```jsx
    return;
```

```jsx
  const dotIndex = filename.lastIndexOf('.');

  if (dotIndex === -1) {
    return '';
  }

  return filename
      .substring(dotIndex)
      .toLowerCase();
}

function getFileError(file) {
  if (!file.type.startsWith('image/')) {
    return '이미지 파일만 선택할 수 있습니다.';
  }

  if (file.size > MAX_FILE_SIZE) {
    return '파일 크기는 5MB 이하만 가능합니다.';
  }

  const extension = getExtension(file.name);

  if (!ALLOWED_EXTENSIONS.includes(extension)) {
    return '지원하지 않는 이미지 형식입니다.';
  }

  return null;
}

function validateFiles(files) {
  attachHelper.textContent = '';

  for (const file of files) {
    const errorMessage = getFileError(file);

    if (errorMessage) {
      attachHelper.textContent = errorMessage;
      return false;
    }
  }

  return true;
}
```

```jsx
      Array.isArray(comments) ? comments : [];

  commentCount.textContent = safeComments.length;

  safeComments.forEach(function (comment) {
    const commentItem =
        createCommentItem(comment);

    commentList.appendChild(commentItem);
  });
}
```
